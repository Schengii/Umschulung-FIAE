// EcoChef Gemini AI Service (TypeScript) mit JSON-Schema Guard
export class GeminiService {
  private apiKey: string;

  constructor(apiKey: string) {
    this.apiKey = apiKey;
  }

  async generateRecipe(ingredients: string[]): Promise<Recipe> {
    const prompt = `Erstelle ein nachhaltiges Rezept mit: ${ingredients.join(', ')}. Antworte strikt in JSON.`;
    const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-pro:generateContent?key=${this.apiKey}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] })
    });
    const result = await response.json();
    return this.validateRecipeSchema(result);
  }
}

// BurgenGame 2D Engine: Asynchroner AI Worker Hook
class AIBot {
  constructor(stateManager, ui) {
    this.stateManager = stateManager;
    this.ui = ui;
    this.initWorker();
  }

  initWorker() {
    this.worker = new Worker('js/workers/ai_worker.js');
    this.worker.postMessage({
      type: 'INIT',
      config: { personality: 'builder', autoAttack: true }
    });

    this.worker.onmessage = (e) => {
      const { action, args, message } = e.data;
      if (typeof this.stateManager[action] === 'function') {
        this.stateManager[action](...args);
        this.ui.showFloatingNotification(message);
      }
    };
  }
}

import { useState, useEffect, useRef } from 'react';

export function useFetch(url, options = {}) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  const cache = useRef({});

  useEffect(() => {
    if (!url) return;
    if (cache.current[url]) {
      setData(cache.current[url]);
      setLoading(false);
      return;
    }

    const controller = new AbortController();
    setLoading(true);

    fetch(url, { ...options, signal: controller.signal })
      .then(res => {
        if (!res.ok) throw new Error(`HTTP error: ${res.status}`);
        return res.json();
      })
      .then(jsonData => {
        cache.current[url] = jsonData;
        setData(jsonData);
      })
      .catch(err => {
        if (err.name !== 'AbortError') setError(err);
      })
      .finally(() => setLoading(false));

    return () => controller.abort();
  }, [url]);

  return { data, loading, error };
}

React Custom Hook (`useFetch`): Implementiert autom. Caching zur Reduzierung von API-Calls, AbortController zur Vermeidung von Race Conditions bei unvollständigen Anfragen und sauberes Laden/Fehler-State-Management.

React Custom Hook (`useFetch`): Implements automatic client-side memory caching, AbortController request cancelation to prevent race conditions on component unmount, and structured loading/error state management.

// Strategy Pattern für Rabattberechnungen
public interface DiscountStrategy {
    double applyDiscount(double price);
}

// Konkrete Implementierung: Prozentualer Rabatt
public class PercentageDiscount implements DiscountStrategy {
    private final double rate;

    public PercentageDiscount(double rate) {
        this.rate = rate;
    }

    @Override
    public double applyDiscount(double price) {
        return price * (1.0 - rate);
    }
}

// Kontext-Klasse (Warenkorb)
public class ShoppingCart {
    private DiscountStrategy strategy;

    public void setDiscountStrategy(DiscountStrategy strategy) {
        this.strategy = strategy;
    }

    public double calculateTotal(double subtotal) {
        if (strategy == null) return subtotal;
        return strategy.applyDiscount(subtotal);
    }
}

Java Strategy Pattern: Demonstriert das Open-Closed-Prinzip (SOLID). Neue Rabattarten können ohne Modifikation des bestehenden Warenkorbs über neue Klassen erweitert werden.

Java Strategy Pattern: Demonstrates the Open-Closed Principle (SOLID). New discount algorithms can be easily added by creating new strategy implementations without changing the shopping cart context class.

// Fetch-Abfrage mit automatischem Offline-Fallback
async function fetchWithFallback(url, fallbackFile) {
  try {
    const response = await fetch(url);
    if (!response.ok) {
      throw new Error(`Server returned status: ${response.status}`);
    }
    return await response.json();
  } catch (error) {
    // Fallback im Offline-Modus
    console.warn("Backend unreachable. Loading offline mock-data...", error);
    const mockRes = await fetch(fallbackFile);
    if (!mockRes.ok) {
      throw new Error("Failed to load mock fallback file.");
    }
    return await mockRes.json();
  }
}

JS Async Fetcher mit Offline Fallback: Sichert die Funktionsfähigkeit der Applikation bei Netzwerkausfällen oder inaktivem Server, indem lokale Mock-Daten geladen werden (Showcase-Modus).

JS Async Fetcher with Offline Fallback: Resolves fetch failures during network timeouts or inactive servers by falling back to local static JSON mock datasets to guarantee application usability.
