// EcoChef Gemini AI Service (TypeScript) mit JSON-Schema Guard
export class GeminiService {
  private apiKey: string;

  constructor(apiKey: string) {
    this.apiKey = apiKey;
  }

  async generateRecipe(ingredients: string[]): Promise<Recipe> {
    const prompt = `Erstelle ein nachhaltiges Rezept mit: ${ingredients.join(', ')}. Antworte strikt in JSON.`;
    const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-pro:generateContent?key=${this.apiKey}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] })
    });
    const result = await response.json();
    return this.validateRecipeSchema(result);
  }
}

// Minecraft C++20 Voxel Engine: Redstone Propagation & Block Updates
struct BlockUpdate {
    glm::ivec3 position;
    uint8_t signalStrength;
};

void RedstoneSimulator::propagateSignal(glm::ivec3 source, uint8_t power) {
    std::queue<BlockUpdate> updateQueue;
    updateQueue.push({source, power});

    while (!updateQueue.empty()) {
        auto current = updateQueue.front();
        updateQueue.pop();

        if (current.signalStrength == 0) continue;

        for (const auto& dir : ADJACENT_DIRECTIONS) {
            glm::ivec3 target = current.position + dir;
            if (world.getBlock(target).type == BlockType::RedstoneWire) {
                uint8_t newPower = current.signalStrength - 1;
                if (newPower > world.getSignal(target)) {
                    world.setSignal(target, newPower);
                    updateQueue.push({target, newPower});
                }
            }
        }
    }
}

import { useState, useEffect, useRef } from 'react';

export function useFetch(url, options = {}) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  const cache = useRef({});

  useEffect(() => {
    if (!url) return;
    if (cache.current[url]) {
      setData(cache.current[url]);
      setLoading(false);
      return;
    }

    const controller = new AbortController();
    setLoading(true);

    fetch(url, { ...options, signal: controller.signal })
      .then(res => {
        if (!res.ok) throw new Error(`HTTP error: ${res.status}`);
        return res.json();
      })
      .then(jsonData => {
        cache.current[url] = jsonData;
        setData(jsonData);
      })
      .catch(err => {
        if (err.name !== 'AbortError') setError(err);
      })
      .finally(() => setLoading(false));

    return () => controller.abort();
  }, [url]);

  return { data, loading, error };
}

React Custom Hook (`useFetch`): Implementiert autom. Caching zur Reduzierung von API-Calls, AbortController zur Vermeidung von Race Conditions bei unvollständigen Anfragen und sauberes Laden/Fehler-State-Management.

React Custom Hook (`useFetch`): Implements automatic client-side memory caching, AbortController request cancelation to prevent race conditions on component unmount, and structured loading/error state management.

// PWA Stale-While-Revalidate Caching mit Offline-Fallback
self.addEventListener('fetch', (event) => {
  if (event.request.method !== 'GET') return;

  event.respondWith(
    caches.open(CACHE_NAME).then(async (cache) => {
      const cachedResponse = await cache.match(event.request);
      const networkFetch = fetch(event.request)
        .then((networkResponse) => {
          if (networkResponse.ok) {
            cache.put(event.request, networkResponse.clone());
          }
          return networkResponse;
        })
        .catch(() => cachedResponse || caches.match('offline.html'));

      return cachedResponse || networkFetch;
    })
  );
});
